cmake_minimum_required(VERSION 3.22)
project(neebles_connect_vnc_prototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
find_package(PkgConfig REQUIRED)
pkg_check_modules(VNCCLIENT REQUIRED IMPORTED_TARGET libvncclient)
add_executable(neebles-connect-vnc main.cpp)
target_link_libraries(neebles-connect-vnc PRIVATE Qt6::Widgets PkgConfig::VNCCLIENT)
target_compile_options(neebles-connect-vnc PRIVATE -Wall -Wextra -Wpedantic)
